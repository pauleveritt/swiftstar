```python
from fastapi.templating import Jinja2Templates
from datetime import datetime

templates = Jinja2Templates(directory="templates")

# In-memory storage for complaints
complaints = []

@app.get("/")
async def home(request: Request):
    return templates.TemplateResponse("home.html", {
        "request": request
    })

@app.get("/complaints")
async def complaints_page(request: Request):
    return templates.TemplateResponse("complaints.html", {
        "request": request,
        "complaints": complaints
    })

    import uvicorn
    uvicorn.run("app:app", reload=True)
```

```python
from dataclasses import dataclass, field
from datetime import datetime, timezone

@dataclass

    agent_name: str
    text: str
    timestamp: datetime = field(default_factory=lambda: datetime.now(timezone.utc))

complaints: list[Complaint] = [

    ),
```

#app.py
```python
from fastapi import FastAPI, Request, Form, RedirectResponse
from fastapi.templating import Jinja2Templates
from fastapi.responses import HTMLResponse
from models import complaints, Complaint

app = FastAPI()
templates = Jinja2Templates(directory="templates")

@app.get("/")
async def home(request: Request):
    return templates.TemplateResponse("home.html", {
        "request": request
    })

@app.get("/complaints")
async def complaints_page(request: Request):
    return templates.TemplateResponse("complaints.html", {
        "request": request,
        "complaints": complaints
    })

@app.post("/complaints")
async def add_complaint(
    request: Request,
    agent_name: str = Form(...),
    text: str = Form(...)
):
    new_complaint = Complaint(agent_name=agent_name, text=text)
    complaints.append(new_complaint)
    return RedirectResponse(url="/complaints", status_code=303)

if __name__ == "__main__":
    import uvicorn
    uvicorn.run("app:app", reload=True)
```
