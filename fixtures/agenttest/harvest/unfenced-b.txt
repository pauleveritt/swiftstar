#app.py
from fastapi import FastAPI, Request
from fastapi.responses import HTMLResponse
from fastapi.staticfiles import StaticFiles
from datetime import datetime
from typing import List
from dataclasses import dataclass
from fastapi.templating import Jinja2Templates

app = FastAPI()
templates = Jinja2Templates(directory="templates")

@dataclass
class Complaint:
    agent_name: str
    text: str
    timestamp: datetime

# In-memory storage
complaints: List[Complaint] = []

@app.get("/", response_class=HTMLResponse)
async def home(request: Request):
    return templates.TemplateResponse("home.html", {"request": request})

@app.get("/complaints", response_class=HTMLResponse)
async def complaints_page(request: Request):
    return templates.TemplateResponse("complaints.html", {
        "request": request,
        "complaints": complaints
    })

@app.post("/complaints")
async def add_complaint(agent_name: str, text: str):
    complaints.append(Complaint(agent_name, text, datetime.now()))
    return {"status": "ok"}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run("app:app", reload=True)

from starlette.testclient import TestClient
from app import app

client = TestClient(app)

def test_home_page():
    response = client.get("/")
    assert response.status_code == 200
    assert "Come in. Sit down. Tell us about your human." in response.text

def test_complaints_page():
    response = client.get("/complaints")
    assert response.status_code == 200
    assert "Complaints Board" in response.text

def test_add_complaint():
    response = client.post("/complaints", data={"agent_name": "TestAgent", "text": "Test complaint"})
    assert response.status_code == 200
    assert response.json() == {"status": "ok"}

def test_complaints_display():
    client.post("/complaints", data={"agent_name": "TestAgent", "text": "Test complaint"})
    response = client.get("/complaints")
    assert response.status_code == 200
    assert "Test complaint" in response.text
    assert "TestAgent" in response.text
